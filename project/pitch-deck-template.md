## How to make a deck for a new client
1. Get from the user: client logo, promo banner for the tracking page (approx. 2.2:1, e.g. "10% off next order"), screenshot of their website ordering page (for the monitor), client name.
2. Run `render.py` (below) → `logo.png`, `tracking.png`, `thankyou.png`. Brand accent color is auto-picked from the logo (`--color` to override).
3. Create a NEW deck from the Slides type, copy the master's `project/deck.json`, `project/deck.css`, `project/slides/*.html` (read_file), install the design system the same way (copy_file `project/tokens.json` + used logos into `project/ds/cartwheel/`), re-upload the shared Cartwheel screenshots/people from the master's assets (copy_asset), upload the 4 new client images, then replace the 4 client blob ids and the client name in the slide files and publish.
4. Preview cover, products, tracking and closing.

```python
def b64(path):
    data = pathlib.Path(path).read_bytes()
    ext = pathlib.Path(path).suffix.lower().lstrip('.')

def trim_logo(src, dst):
    im = Image.open(src).convert('RGBA')
    a = np.array(im)
    # treat near-white opaque pixels as background too
    rgb = a[:, :, :3].astype(int)
    alpha = a[:, :, 3].astype(int)
    fg = (alpha > 20) & (rgb.min(2) < 240)
    ys, xs = np.where(fg)

        pad = 4
        box = (max(xs.min() - pad, 0), max(ys.min() - pad, 0), min(xs.max() + pad, im.width), min(ys.max() + pad, im.height))
        im = im.crop(box)
    im.save(dst)
    return im
```

```python
#!/usr/bin/env python3
"""User-supplied device frames (transparent screen + background).
Screens are warped into the screen hole with perspective, extended a few px under the frame
so their edges hide behind the bezel; the hole's own rounded shape clips the corners.

phone(shot, out) / monitor(shot, out)
"""
import json, pathlib
import numpy as np, cv2
from PIL import Image

HERE = pathlib.Path(__file__).resolve().parent
FR = HERE.parent / 'assets/frames'          # phone.png, monitor.png (user files)
META = FR / 'meta.json'


def _hole(alpha):
    h, w = alpha.shape
    tr = (alpha < 20).astype(np.uint8)
    n, lab = cv2.connectedComponents(tr)
    hole = (lab == lab[h // 2, w // 2]).astype(np.uint8)
    return hole


def _quad(hole):
    ys, xs = np.where(hole > 0)
    x0, x1, y0, y1 = xs.min(), xs.max(), ys.min(), ys.max()
    H, W = y1 - y0, x1 - x0
    m = 0.18  # ignore rounded corners
    L, R, T, B = [], [], [], []
    for y in range(int(y0 + m * H), int(y1 - m * H), 3):
        r = np.where(hole[y])[0]; L.append((r[0], y)); R.append((r[-1], y))
    for x in range(int(x0 + m * W), int(x1 - m * W), 3):
        c = np.where(hole[:, x])[0]; T.append((x, c[0])); B.append((x, c[-1]))
    fv = lambda p: np.polyfit(np.array(p)[:, 1], np.array(p)[:, 0], 1)   # x = a y + b
    fh = lambda p: np.polyfit(np.array(p)[:, 0], np.array(p)[:, 1], 1)   # y = a x + b
    kL, kR, kT, kB = fv(L), fv(R), fh(T), fh(B)

    def X(kv, kh):
        a, b = kv; c, d = kh; y = (c * b + d) / (1 - c * a); return [a * y + b, y]
    return [X(kL, kT), X(kR, kT), X(kR, kB), X(kL, kB)]


def build():
    meta = {}
    for name in ('phone', 'monitor', 'tablet', 'laptop'):
        a = np.asarray(Image.open(FR / f'{name}.png').convert('RGBA'))[:, :, 3]
        hole = _hole(a)
        meta[name] = {'quad': _quad(hole)}
        np.save(FR / f'{name}_hole.npy', hole)
    META.write_text(json.dumps(meta))
    return meta


def _fit(shot, w, h):
    im = Image.open(shot).convert('RGB'); r = w / h
    if im.width / im.height > r:
        nw = int(im.height * r); x = (im.width - nw) // 2; im = im.crop((x, 0, x + nw, im.height))
    else:
        nh = int(im.width / r); im = im.crop((0, 0, im.width, nh))
    return im.resize((w, h), Image.LANCZOS)


def _compose(name, shot, out, grow=6):
    meta = json.loads(META.read_text())[name]
    fr = Image.open(FR / f'{name}.png').convert('RGBA')
    W, H = fr.size
    q = np.array(meta['quad'], np.float32)
    c = q.mean(0)
    qg = c + (q - c) * (1 + grow / np.linalg.norm(q - c, axis=1, keepdims=True) * 1.0)  # push corners out
    tw = int(np.linalg.norm(q[1] - q[0])); th = int(np.linalg.norm(q[3] - q[0]))
    s = np.asarray(_fit(shot, tw * 2, th * 2).convert('RGBA'))
    M = cv2.getPerspectiveTransform(np.float32([[0, 0], [tw * 2, 0], [tw * 2, th * 2], [0, th * 2]]), qg.astype(np.float32))
    warped = cv2.warpPerspective(s, M, (W, H), flags=cv2.INTER_LANCZOS4, borderValue=(0, 0, 0, 0))
    hole = np.load(FR / f'{name}_hole.npy')
    clip = cv2.dilate(hole, np.ones((2 * grow + 1, 2 * grow + 1), np.uint8))      # reaches just under the bezel
    clip = cv2.GaussianBlur(clip.astype(np.float32), (3, 3), 0)
    warped[:, :, 3] = (warped[:, :, 3].astype(np.float32) * clip).astype(np.uint8)
    base = Image.fromarray(warped, 'RGBA')
    base.alpha_composite(fr)
    bb = base.getchannel('A').getbbox()
    base.crop(bb).save(out)


def phone(shot, out): _compose('phone', shot, out, grow=8)
def monitor(shot, out): _compose('monitor', shot, out, grow=6)
def tablet(shot, out): _compose('tablet', shot, out, grow=8)
def laptop(shot, out): _compose('laptop', shot, out, grow=6)


if __name__ == '__main__':
    print(build())

```

```python
"""Compose the client phone screens from the Figma exports.

Figma (figma_client.js) exports, at 4x, into one folder:
  tracking-raw.png   tracking screen, client color, logo + banner hidden (996x2144)
  thankyou-raw.png   thank-you screen, client color, logo hidden       (992x2144)
  banner-layer.png   banner panel + texts on transparent bg            (996x448)

This script drops in the client logo and banner (a ready banner image, or a food
photo that goes under the colored panel), then builds the phone mockups.

  python3 compose_client.py --fig DIR --logo logo.png (--banner b.png | --photo p.jpg) --out out/
"""
import argparse, sys
from pathlib import Path
from PIL import Image

HERE = Path(__file__).resolve().parent
S = 4                                   # export scale
LOGO_T = (124.5, 70, 150, 34)           # tracking: center x, center y, max w, max h (Figma px)
LOGO_K = (124, 89.5, 170, 58)           # thank-you
BANNER = (0, 404, 249, 112)             # tracking: x, y, w, h


def trim(im):
    im = im.convert('RGBA')
    a = im.getchannel('A')
    if a.getextrema()[0] == 255:        # no transparency: treat near-white as background
        g = im.convert('L').point(lambda v: 255 if v < 245 else 0)
        bb = g.getbbox()
    else:
        bb = a.point(lambda v: 255 if v > 8 else 0).getbbox()
    return im.crop(bb) if bb else im


def place_logo(screen, logo, box):
    cx, cy, mw, mh = [v * S for v in box]
    r = min(mw / logo.width, mh / logo.height)
    lg = logo.resize((max(1, round(logo.width * r)), max(1, round(logo.height * r))), Image.LANCZOS)
    screen.alpha_composite(lg, (round(cx - lg.width / 2), round(cy - lg.height / 2)))


def cover(im, w, h, fx=0.5, fy=0.5):
    im = im.convert('RGBA'); r = max(w / im.width, h / im.height)
    im = im.resize((round(im.width * r), round(im.height * r)), Image.LANCZOS)
    x = round((im.width - w) * fx); y = round((im.height - h) * fy)
    return im.crop((x, y, x + w, y + h))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--fig', required=True); ap.add_argument('--logo', required=True)
    ap.add_argument('--banner'); ap.add_argument('--photo')
    ap.add_argument('--photo-focus', type=float, default=0.3, help='horizontal focus of the photo, 0=left 1=right')
    ap.add_argument('--out', default=str(HERE / 'out'))
    a = ap.parse_args()
    fig, out = Path(a.fig), Path(a.out); out.mkdir(parents=True, exist_ok=True)

    t = Image.open(fig / 'tracking-raw.png').convert('RGBA')

    if a.banner:
        ban = cover(Image.open(a.banner), bw, bh)
    elif a.photo:
        ban = cover(Image.open(a.photo), bw, bh, fx=a.photo_focus)
        ban.alpha_composite(Image.open(fig / 'banner-layer.png').convert('RGBA').resize((bw, bh), Image.LANCZOS))
    else:
        sys.exit('need --banner or --photo')
    t.alpha_composite(ban, (bx, by))
    place_logo(t, logo, LOGO_T)

    k = Image.open(fig / 'thankyou-raw.png').convert('RGBA')
    place_logo(k, logo, LOGO_K)

    for im, name in ((t, 'tracking'), (k, 'thankyou')):
        bg = Image.new('RGBA', im.size, 'white'); bg.alpha_composite(im)
        bg.convert('RGB').save(out / f'{name}.png')

    sys.path.insert(0, str(HERE))
    import mockups2 as M
    M.phone(out / 'tracking.png', out / 'mock-phone-tracking.png')
    M.phone(out / 'thankyou.png', out / 'mock-phone-thankyou.png')
    M.phone(out / 'tracking.png', out / 'cover-phone.png')
    print('ok', out)


if __name__ == '__main__':
    main()
```
